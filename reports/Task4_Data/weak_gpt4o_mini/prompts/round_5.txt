    public static String parseToken(final String str, final char[] terminators) {
        if (str == null || terminators == null) {
            return str;
        }
        for (int i = 0; i < str.length(); i++) {
            for (char terminator : terminators) {

                }
            }
        }
        return str;
    }

    private static boolean isOneOf(char ch, final char[] charray) {
        for (char c : charray) {
            if (c == ch) {

            }
        }
        return false;
    }

    public static int extractIntInStr(String str) {
        if (str == null || str.isEmpty()) {
            return 0;
        }
        StringBuilder digits = new StringBuilder();
        for (int i = str.length() - 1; i >= 0; i--) {
            char ch = str.charAt(i);
            if (Character.isDigit(ch)) {
                digits.append(ch);
            } else if (digits.length() > 0) {
                break;
            }
        }
        return digits.length() > 0 ? Integer.parseInt(digits.reverse().toString()) : 0;
    }
